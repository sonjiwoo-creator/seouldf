# %% 필터링된 지도 생성 (상권유형/분류별)
import folium

def create_filtered_map(filtered_data, title, color_func, output_filename):
    """필터링된 데이터를 바탕으로 지도 생성"""
    seoul_center = [37.5665, 126.9780]
    m = folium.Map(location=seoul_center, zoom_start=11, tiles='cartodbpositron')
    
    for idx, row in filtered_data.iterrows():
        color = color_func(row)
        popup_text = f"""
        <b>{row['상권_코드_명']}</b><br>
        자치구: {row['자치구_코드_명']}<br>
        상권유형: {row['상권_구분_코드_명']}<br>
        분류: {row['상권분류']}<br>
        성장유형: {row['성장유형']}<br>
        RFM합계: {row['RFM_합계']}<br>
        매출증감률: {row['R_매출증감률']:.1%}<br>
        객단가: {row['M_객단가']:,.0f}원
        """
        folium.CircleMarker(
            location=[row['lat'], row['lon']],
            radius=6,
            popup=folium.Popup(popup_text, max_width=300, parse_html=False),
            color=color,
            fill=True,
            fillColor=color,
            fillOpacity=0.7,
            weight=2
        ).add_to(m)
    
    m.save(str(OUTPUT_DIR / output_filename))
    print(f'✅ 지도 저장: {output_filename} ({len(filtered_data)}개 상권)')
